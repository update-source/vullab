# USER MODULE
user
----
id PK int
username varchar(50) unique
email varchar(100) unique
password varchar(255)
role varchar(20)
is_active bool
is_email_verified bool
last_login_at timestamp
created_at timestamp
updated_at timestamp

user_token
----
id PK int
user_id int FK >- user.id
token_type varchar(50)
token_value varchar(255)
expires_at timestamp
is_used bool
created_at timestamp

user_profile
----
id PK int
user_id int unique FK - user.id
first_name varchar(50)
last_name varchar(50)
phone_number varchar(20)
date_of_birth date
avatar_url varchar(255)
bio text
created_at timestamp
updated_at timestamp

# ADDRESS MODULE
country
----
id PK int
country_name varchar(100) unique
country_code varchar(3) unique

address
----
id PK int
street_address varchar(255)
city varchar(100)
postal_code varchar(20)
country_id int FK >- country.id
created_at timestamp

user_address
----
id PK int
user_id int FK >- user.id
address_id int FK >- address.id
is_default bool
address_type varchar(20)

# PRODUCT MODULE
category
----
id PK int
parent_category_id int FK >- category.id
category_name varchar(100)
slug varchar(100) unique
is_active bool
created_at timestamp

product
----
id PK int
category_id int FK >- category.id
name varchar(255)
description text
sku varchar(50) unique
is_published bool
created_at timestamp

product_item
----
id PK int
product_id int FK >- product.id
sku varchar(50) unique
price decimal(10,2)
quantity_in_stock int
is_active bool
created_at timestamp

user_review
----
id PK int
user_id int FK >- user.id
product_id int FK >- product.id
order_id int FK >- shop_order.id
rating int
title varchar(200)
comment text
created_at timestamp

# CART MODULE
shopping_cart
----
id PK int
user_id int unique FK - user.id
expires_at timestamp
created_at timestamp

shopping_cart_item
----
id PK int
cart_id int FK >- shopping_cart.id
product_item_id int FK >- product_item.id
quantity int
created_at timestamp

# ORDER MODULE
order_status
----
id PK int
code varchar(50) unique
display_name varchar(100)

shipping_method
----
id PK int
code varchar(50) unique
name varchar(100)
price decimal(10,2)

payment_method
----
id PK int
code varchar(50) unique
method_name varchar(100)

shop_order
----
id PK int
user_id int FK >- user.id
order_number varchar(50) unique
subtotal decimal(10,2)
shipping_cost decimal(10,2)
tax_amount decimal(10,2)
discount_amount decimal(10,2)
final_total decimal(10,2)
shipping_address_id int FK >- address.id
shipping_method_id int FK >- shipping_method.id
payment_method_id int FK >- payment_method.id
order_status_id int FK >- order_status.id
created_at timestamp

order_line
----
id PK int
order_id int FK >- shop_order.id
product_item_id int FK >- product_item.id
quantity int
unit_price decimal(10,2)
total_price decimal(10,2)
product_name varchar(255)
product_sku varchar(50)

# PROMOTION MODULE
promotion
----
id PK int
code varchar(50) unique
name varchar(100)
discount_type varchar(20)
discount_value decimal(10,2)
usage_limit int
is_active bool
created_at timestamp

promotion_category
----
promotion_id int PK FK >- promotion.id
category_id int PK FK >- category.id
